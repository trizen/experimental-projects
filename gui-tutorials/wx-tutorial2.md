Here is a complete, runnable example showing a `wxGlade` `.wxg` layout file, the generated Perl UI module (`DemoUI.pm`), and a subclassed Perl application (`app.pl`) that wires up a **menu bar**, **status bar**, and **modal dialog** (`wxDialog`).

### 1. Project Structure

Place all three files in the same directory:

* `demo_app.wxg` — `wxGlade` project file (can be opened directly in `wxGlade` and regenerated to `DemoUI.pm`).
* `DemoUI.pm` — Generated UI classes (`DemoMainFrame` and `DemoPrefsDialog`).
* `app.pl` — Application entry point and event/dialog logic.

### 3. Generated UI Module (`DemoUI.pm`)

This module defines `DemoMainFrame` (with menu bar, two-field status bar, and panel widgets) and `DemoPrefsDialog` (a modal dialog with standard OK/Cancel buttons).

```perl
# generated by wxGlade

package DemoMainFrame;

use Wx qw[:everything];
use base qw(Wx::Frame);
use strict;

sub new {
    my ($self, $parent, $id, $title, $pos, $size, $style, $name) = @_;
    $parent = undef              unless defined $parent;
    $id     = -1                 unless defined $id;
    $title  = ""                 unless defined $title;
    $pos    = wxDefaultPosition  unless defined $pos;
    $size   = wxDefaultSize      unless defined $size;
    $style  = wxDEFAULT_FRAME_STYLE unless defined $style;
    $name   = ""                 unless defined $name;

    $style = wxDEFAULT_FRAME_STYLE;
    $size = Wx::Size->new(520, 320);

    $self = $self->SUPER::new($parent, $id, $title, $pos, $size, $style, $name);
    $self->SetTitle("Perl Wx Menu, Status & Dialog Demo");

    # Menu Bar
    $self->{main_frame_menubar} = Wx::MenuBar->new();
    my $wxglade_tmp_menu;
    $wxglade_tmp_menu = Wx::Menu->new();
    $self->{menu_preferences} = $wxglade_tmp_menu->Append(wxID_ANY, "&Preferences...\tCtrl+P", "");
    $wxglade_tmp_menu->AppendSeparator();
    $self->{menu_exit} = $wxglade_tmp_menu->Append(wxID_ANY, "E&xit\tCtrl+Q", "");
    $self->{main_frame_menubar}->Append($wxglade_tmp_menu, "&File");

    $wxglade_tmp_menu = Wx::Menu->new();
    $self->{menu_about} = $wxglade_tmp_menu->Append(wxID_ANY, "&About", "");
    $self->{main_frame_menubar}->Append($wxglade_tmp_menu, "&Help");

    $self->SetMenuBar($self->{main_frame_menubar});

    # Status Bar
    $self->{main_frame_statusbar} = $self->CreateStatusBar(2);
    $self->{main_frame_statusbar}->SetStatusWidths(-1, 180);
    $self->{main_frame_statusbar}->SetStatusText("Ready", 0);
    $self->{main_frame_statusbar}->SetStatusText("User: Guest", 1);

    my $sizer_root = Wx::BoxSizer->new(wxVERTICAL);

    $self->{panel_main} = Wx::Panel->new($self, wxID_ANY);
    $sizer_root->Add($self->{panel_main}, 1, wxALL|wxEXPAND, 12);

    my $sizer_panel = Wx::BoxSizer->new(wxVERTICAL);

    $self->{lbl_summary} = Wx::StaticText->new($self->{panel_main}, wxID_ANY, "Current profile: Guest | Auto-save: Off");
    $sizer_panel->Add($self->{lbl_summary}, 0, wxALL|wxEXPAND, 4);

    $self->{txt_notes} = Wx::TextCtrl->new($self->{panel_main}, wxID_ANY, "", wxDefaultPosition, wxDefaultSize, wxTE_MULTILINE);
    $sizer_panel->Add($self->{txt_notes}, 1, wxALL|wxEXPAND, 8);

    $self->{btn_edit_prefs} = Wx::Button->new($self->{panel_main}, wxID_ANY, "Edit Preferences...");
    $sizer_panel->Add($self->{btn_edit_prefs}, 0, wxALL|wxALIGN_RIGHT, 4);

    $self->{panel_main}->SetSizer($sizer_panel);
    $self->SetSizer($sizer_root);
    $self->Layout();

    Wx::Event::EVT_MENU($self, $self->{menu_preferences}->GetId, $self->can('on_menu_preferences'));
    Wx::Event::EVT_MENU($self, $self->{menu_exit}->GetId, $self->can('on_menu_exit'));
    Wx::Event::EVT_MENU($self, $self->{menu_about}->GetId, $self->can('on_menu_about'));
    Wx::Event::EVT_BUTTON($self, $self->{btn_edit_prefs}->GetId, $self->can('on_open_preferences_btn'));

    return $self;
}

package DemoPrefsDialog;

use Wx qw[:everything];
use base qw(Wx::Dialog);
use strict;

sub new {
    my ($self, $parent, $id, $title, $pos, $size, $style, $name) = @_;
    $parent = undef              unless defined $parent;
    $id     = -1                 unless defined $id;
    $title  = ""                 unless defined $title;
    $pos    = wxDefaultPosition  unless defined $pos;
    $size   = wxDefaultSize      unless defined $size;
    $style  = wxDEFAULT_DIALOG_STYLE unless defined $style;
    $name   = ""                 unless defined $name;

    $style = wxDEFAULT_DIALOG_STYLE|wxRESIZE_BORDER;
    $size = Wx::Size->new(360, 200);

    $self = $self->SUPER::new($parent, $id, $title, $pos, $size, $style, $name);
    $self->SetTitle("Preferences");

    my $sizer_dlg_root = Wx::BoxSizer->new(wxVERTICAL);

    $self->{lbl_username} = Wx::StaticText->new($self, wxID_ANY, "Display Name:");
    $sizer_dlg_root->Add($self->{lbl_username}, 0, wxLEFT|wxRIGHT|wxTOP|wxEXPAND, 12);

    $self->{txt_username} = Wx::TextCtrl->new($self, wxID_ANY, "");
    $sizer_dlg_root->Add($self->{txt_username}, 0, wxLEFT|wxRIGHT|wxTOP|wxEXPAND, 12);

    $self->{chk_autosave} = Wx::CheckBox->new($self, wxID_ANY, "Enable auto-save status indicator");
    $sizer_dlg_root->Add($self->{chk_autosave}, 0, wxALL|wxEXPAND, 12);

    $sizer_dlg_root->Add(1, 1, 1, wxEXPAND, 0);

    my $sizer_dlg_buttons = Wx::StdDialogButtonSizer->new();
    $sizer_dlg_root->Add($sizer_dlg_buttons, 0, wxALL|wxALIGN_RIGHT, 12);

    $self->{button_OK} = Wx::Button->new($self, wxID_OK, "");
    $self->{button_OK}->SetDefault();
    $sizer_dlg_buttons->Add($self->{button_OK}, 1, wxEXPAND, 0);

    $self->{button_CANCEL} = Wx::Button->new($self, wxID_CANCEL, "");
    $sizer_dlg_buttons->Add($self->{button_CANCEL}, 1, wxEXPAND, 0);

    $sizer_dlg_buttons->Realize();

    $self->SetSizer($sizer_dlg_root);
    $self->Layout();

    return $self;
}

1;
```

### 4. Application Logic (`app.pl`)

This file subclasses `DemoMainFrame`, implements the menu and button callbacks, launches `DemoPrefsDialog` modally with `ShowModal()`, updates the status bar and summary label only when `wxID_OK` is returned, and explicitly destroys the dialog afterward.

```perl
#!/usr/bin/env perl
use strict;
use warnings;
use lib '.';
use Wx qw[:everything];
use DemoUI;

package MainAppFrame;
use base qw(DemoMainFrame);

sub new {
    my ($class, $parent) = @_;
    my $self = $class->SUPER::new($parent);

    $self->{username} = 'Guest';
    $self->{autosave} = 0;
    $self->_refresh_ui();

    return $self;
}

sub _refresh_ui {
    my ($self) = @_;
    my $save_text = $self->{autosave} ? 'On' : 'Off';

    $self->{lbl_summary}->SetLabel(
        "Current profile: $self->{username} | Auto-save: $save_text"
    );
    $self->{main_frame_statusbar}->SetStatusText("User: $self->{username}", 1);
    $self->{panel_main}->Layout();
}

sub on_open_preferences_btn {
    my ($self, $event) = @_;
    $self->on_menu_preferences($event);
}

sub on_menu_preferences {
    my ($self, $event) = @_;

    my $dlg = DemoPrefsDialog->new($self);
    $dlg->{txt_username}->SetValue($self->{username});
    $dlg->{chk_autosave}->SetValue($self->{autosave} ? 1 : 0);

    $self->{main_frame_statusbar}->SetStatusText("Editing preferences...", 0);

    my $rc = $dlg->ShowModal();
    if ($rc == wxID_OK) {
        my $name = $dlg->{txt_username}->GetValue;
        $name =~ s/^\s+|\s+$//g;
        $self->{username} = length($name) ? $name : 'Guest';
        $self->{autosave} = $dlg->{chk_autosave}->GetValue ? 1 : 0;

        $self->_refresh_ui();
        $self->{main_frame_statusbar}->SetStatusText("Preferences saved", 0);
    }
    else {
        $self->{main_frame_statusbar}->SetStatusText("Preferences cancelled", 0);
    }

    $dlg->Destroy();
}

sub on_menu_about {
    my ($self, $event) = @_;
    Wx::MessageBox(
        "Perl + Wx + wxGlade Example\nMenu bar, two-field status bar, and modal dialog.",
        "About",
        wxOK | wxICON_INFORMATION,
        $self
    );
}

sub on_menu_exit {
    my ($self, $event) = @_;
    $self->Close(1);
}

package DemoWxApp;
use base qw(Wx::App);

sub OnInit {
    my ($self) = @_;
    my $frame = MainAppFrame->new(undef);
    $self->SetTopWindow($frame);
    $frame->Show(1);
    return 1;
}

package main;
DemoWxApp->new->MainLoop;

```

### Key Mechanics Illustrated Here

* **Status bar fields:** `CreateStatusBar(2)` creates two panes—field `0` (flexible width `-1`) shows transient status messages, and field `1` (fixed width `180`) shows the active user.
* **Standard dialog IDs:** Using `wxID_OK` and `wxID_CANCEL` on the dialog buttons lets `wxWidgets` automatically close the modal loop and return `wxID_OK` or `wxID_CANCEL` from `ShowModal()` without extra button handlers.
* **Lifecycle cleanup:** Calling `$dlg->Destroy()` after `ShowModal()` frees the native C++ window handle cleanly on Windows.
